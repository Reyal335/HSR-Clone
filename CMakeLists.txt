cmake_minimum_required(VERSION 3.15)
project(MyProject)

set(CMAKE_CXX_STANDARD 17)
file(GLOB_RECURSE PROJECT_SOURCES CONFIGURE_DEPENDS "src/*.cpp")
add_executable(MyProject ${PROJECT_SOURCES})
target_include_directories(MyProject PRIVATE src)

if(MINGW)
    target_link_libraries(MyProject PRIVATE ncursesw)
else()
    find_package(Curses REQUIRED)
    target_include_directories(MyProject PRIVATE ${CURSES_INCLUDE_DIRS})
    target_link_libraries(MyProject PRIVATE ${CURSES_LIBRARIES})
endif()
